Using the hooks
const mon = new PPG.PulseMonitor({ video, durationSec: 60 });
mon.on('rr', r => …)        // {t, rr, valid, reason, bpm}
mon.on('metrics', m => …)   // {bpm, rmssd, sdnn, hrvReliable, hrvIssue, progress}
mon.on('quality', q => …)   // {score, level, hint}
mon.on('complete', res => …)
await mon.start();
